

# Конфигурационный файл проекта
```
{
    // Параметры физической модели
    "model": {
        // Радиус окружности, по которой движется объект (метр)
        "radius": 1.0,
        // Ускорение свободного падения (метр/секунда^2)
        "gravity": 9.81,
        // Коэффициент вязкого трения (ньютон*секунда/метр)
        "friction_coeff": 0.2,
        // Масса движущегося объекта (килограмм)
        "mass": 1.0
    },
    // Начальные условия для симуляции
    "initial_conditions": {
        // Начальный угол отклонения в радианах (положительный против часовой стрелки от оси X)
        // Объект начинает движение в точке (R*sin(angle_rad), -R*cos(angle_rad))
        "angle_rad": 0.1,
        // Начальная проекция скорости на ось X (метр/секунда)
        "vx0": 0.0,
        // Начальная проекция скорости на ось Y (метр/секунда)
        // В данной модели ось Y направлена вниз, поэтому отрицательное значение означало бы
        // начальную скорость вверх.
        "vy0": 0.0
    },
    // Параметры численного интегрирования
    "simulation": {
        // Начальное время симуляции (секунда)
        "t_start": 0.0,
        // Конечное время симуляции (секунда)
        "t_end": 5.0,
        // Большой шаг интегрирования для методов Эйлера, Хьюна и RK4
        // Используется для демонстрации различий между методами
        "dt_coarse": 0.05,
        // Малый шаг интегрирования для метода Рунге-Кутты 4-го порядка
        // Используется как "эталонное" решение для сравнения
        "dt_fine": 0.005
    },
    // Параметры анимации
    "animation": {
        // Флаг, определяющий, следует ли сохранять анимацию в файл GIF
        "save_gif": false,
        // Имя файла для сохранения анимации (если save_gif=true)
        "gif_filename": "animation.gif",
        // Частота кадров анимации (кадры в секунду)
        "fps": 30
    }
}
```
